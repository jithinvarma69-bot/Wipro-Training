Bookstore Swagger API – CRUD Operations with Test Scripts 

1. GET Request – Fetch All Books
Endpoint:
GET /books
Test Script:
pm.test("Status code is 200", function () {
    pm.response.to.have.status(200);
});

pm.test("Response is in JSON format", function () {
    pm.response.to.be.json;
});

pm.test("Response is an array", function () {
    let jsonData = pm.response.json();
    pm.expect(jsonData).to.be.an("array");
});

Response (Sample):
[
  {
    "id": 1,
    "title": "Book One",
    "author": "Author A",
    "price": 300
  }
]

2. POST Request – Create a New Book
Endpoint:
POST /books
Request Body:
{
  "title": "Atomic Habits",
  "author": "James Clear",
  "price": 499
}
Test Script:
pm.test("Status code is 201", function () {
    pm.response.to.have.status(201);
});
pm.test("Book created successfully", function () {
    let jsonData = pm.response.json();
    pm.expect(jsonData.title).to.eql("Atomic Habits");
});
// Save ID for further requests
pm.environment.set("book_id", pm.response.json().id);
Response (Sample):
{
  "id": 101,
  "title": "Atomic Habits",
  "author": "James Clear",
  "price": 499
}
