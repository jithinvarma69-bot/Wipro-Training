3. Admin User Search (Data Tables) Requirement: Write a feature for searching existing users in the "Admin" module. Scenario: Filter users based on specific criteria. Challenge: Use a Gherkin Data Table (Step Arguments) within a single `When` step to input multiple search parameters at once (e.g., Username, User Role, and Status).

Code:

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import io.cucumber.java.en.*;
import java.util.List;
import java.util.Map;
import static org.testng.Assert.assertTrue;
public class SearchUserSteps {
    private WebDriver driver;
    @Given("the user is logged in with username {string} and password {string}")
    public void login(String username, String password) {
        driver = new ChromeDriver();
        driver.get("https://opensource-demo.orangehrmlive.com/web/index.php/auth/login");
        driver.findElement(By.name("username")).sendKeys(username);
        driver.findElement(By.name("password")).sendKeys(password);
        driver.findElement(By.xpath("//button[@type='submit']")).click();
    }
    @When("the user navigates to the Admin module")
    public void goToAdminModule() {
        driver.findElement(By.xpath("//span[text()='Admin']")).click();
    }
    @When("enters the following search criteria:")
    public void enterSearchCriteria(io.cucumber.datatable.DataTable dataTable) {
        List<Map<String, String>> criteria = dataTable.asMaps(String.class, String.class);
        for (Map<String, String> row : criteria) {
            String username = row.get("Username");
            String role = row.get("User Role");
            String status = row.get("Status");
            driver.findElement(By.xpath("//label[text()='Username']/following::input[1]")).clear();
            driver.findElement(By.xpath("//label[text()='Username']/following::input[1]")).sendKeys(username);
            WebElement roleDropdown = driver.findElement(By.xpath("//label[text()='User Role']/following::div[@role='listbox'][1]"));
            roleDropdown.click();
            driver.findElement(By.xpath("//div[@role='option']//span[text()='" + role + "']")).click();
            WebElement statusDropdown = driver.findElement(By.xpath("//label[text()='Status']/following::div[@role='listbox'][1]"));
            statusDropdown.click();
            driver.findElement(By.xpath("//div[@role='option']//span[text()='" + status + "']")).click();
        }
    }
    @When("clicks the Search button")
    public void clickSearch() {
        driver.findElement(By.xpath("//button[text()=' Search ']")).click();
    }
    @Then("the filtered results should display the user {string}")
    public void verifyUserInResults(String expectedUser) {
        List<WebElement> users = driver.findElements(By.xpath("//div[@class='oxd-table-body']/div/div[2]"));
        boolean found = users.stream().anyMatch(u -> u.getText().equalsIgnoreCase(expectedUser));
        assertTrue(found, "User not found: " + expectedUser);
        driver.quit();
    }
}

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import io.cucumber.java.en.*;
import java.util.List;
import java.util.Map;
import static org.testng.Assert.assertEquals;
import static org.testng.Assert.assertTrue;
public class LeaveApplicationSteps {
    private WebDriver driver;
    private int initialBalance;
    @Given("the user is logged in with username {string} and password {string}")
    public void login(String username, String password) {
        driver = new ChromeDriver();
        driver.get("https://opensource-demo.orangehrmlive.com/web/index.php/auth/login");
        driver.findElement(By.name("username")).sendKeys(username);
        driver.findElement(By.name("password")).sendKeys(password);
        driver.findElement(By.xpath("//button[@type='submit']")).click();
    }
    @When("the user navigates to the Leave module")
    public void goToLeaveModule() {
        driver.findElement(By.xpath("//span[text()='Leave']")).click();
    }
    @When("clicks \"Apply Leave\"")
    public void clickApplyLeave() {
        driver.findElement(By.xpath("//button[text()=' Apply ']")).click();
    }
    @When("selects \"Medical Leave\" from the leave type dropdown")
    public void selectMedicalLeave() {
        WebElement leaveTypeDropdown = driver.findElement(By.xpath("//label[text()='Type']/following::div[@role='listbox'][1]"));
        leaveTypeDropdown.click();
        driver.findElement(By.xpath("//div[@role='option']//span[text()='Medical Leave']")).click();
        // capture initial balance
        WebElement balanceElement = driver.findElement(By.xpath("//label[text()='Leave Balance']/following-sibling::div"));
        initialBalance = Integer.parseInt(balanceElement.getText());
    }
    @When("enters leave details:")
    public void enterLeaveDetails(io.cucumber.datatable.DataTable dataTable) {
        List<Map<String, String>> details = dataTable.asMaps(String.class, String.class);
        for (Map<String, String> row : details) {
            driver.findElement(By.name("fromDate")).sendKeys(row.get("From Date"));
            driver.findElement(By.name("toDate")).sendKeys(row.get("To Date"));
            driver.findElement(By.xpath("//textarea")).sendKeys(row.get("Comment"));
        }
    }
    @When("clicks the \"Submit\" button")
    public void clickSubmit() {
        driver.findElement(By.xpath("//button[text()=' Submit ']")).click();
    }
    @Then("a success message {string} should appear")
    public void verifySuccessMessage(String expectedMessage) throws InterruptedException {
        Thread.sleep(1000); // wait for toast
        String actualMessage = driver.findElement(By.xpath("//p[contains(@class,'oxd-text--toast-message')]")).getText();
        assertEquals(actualMessage, expectedMessage);
    }
    @Then("the leave balance for \"Medical Leave\" should decrease accordingly")
    public void verifyLeaveBalanceDecrease() {
        WebElement balanceElement = driver.findElement(By.xpath("//label[text()='Leave Balance']/following-sibling::div"));
        int updatedBalance = Integer.parseInt(balanceElement.getText());
        assertTrue(updatedBalance < initialBalance, "Leave balance did not decrease");
    }
    @Then("the leave request status should show \"Pending Approval\"")
    public void verifyLeaveRequestStatus() {
        String status = driver.findElement(By.xpath("//div[@class='oxd-table-body']/div/div[6]")).getText();
        assertEquals(status, "Pending Approval");
        driver.quit();
    }
}
