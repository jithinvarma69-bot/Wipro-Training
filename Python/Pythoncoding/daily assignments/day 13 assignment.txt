package stepDefinitions;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver;
import io.cucumber.java.en.*;
import org.junit.Assert;
public class LoginSteps {
    WebDriver driver;
    // Given step: Initialize driver and navigate to URL
    @Given("user is on the OrangeHRM login page")
    public void user_is_on_login_page() {
        // Set path to chromedriver if needed
        System.setProperty("webdriver.chrome.driver", "path/to/chromedriver");
        driver = new ChromeDriver();
        driver.manage().window().maximize();
        driver.get("https://opensource-demo.orangehrmlive.com/web/index.php/auth/login");
    }
    // When step: Enter username and password (reusable)
    @When("user enters username {string} and password {string}")
    public void user_enters_username_and_password(String username, String password) {
        driver.findElement(By.name("username")).sendKeys(username);
        driver.findElement(By.name("password")).sendKeys(password);
        driver.findElement(By.xpath("//button[@type='submit']")).click();
    }
    // Then step: Verify successful login using URL assertion
    @Then("user should be redirected to dashboard")
    public void user_should_be_redirected_to_dashboard() {
        // Wait for page to load (simple implicit wait alternative)
        try {
            Thread.sleep(2000); // Not best practice; use WebDriverWait in real projects
        } catch (InterruptedException e) {
            e.printStackTrace();
        }
        String currentUrl = driver.getCurrentUrl();
        System.out.println("Current URL after login: " + currentUrl);
        Assert.assertTrue("Dashboard not loaded!", currentUrl.contains("dashboard"));
        
        // Close the browser after assertion
        driver.quit();
    }
}

package stepDefinitions;

import io.cucumber.datatable.DataTable;
import io.cucumber.java.en.*;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.Select;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;

import java.time.Duration;
import java.util.List;
import java.util.Map;

public class AdminSteps {

    WebDriver driver;
    WebDriverWait wait;

    @Given("user is logged in and on the Admin page")
    public void user_is_logged_in_and_on_admin_page() {
        System.setProperty("webdriver.chrome.driver", "path/to/chromedriver");
        driver = new ChromeDriver();
        driver.manage().window().maximize();

        driver.get("https://opensource-demo.orangehrmlive.com/web/index.php/auth/login");
        driver.findElement(By.name("username")).sendKeys("Admin");
        driver.findElement(By.name("password")).sendKeys("admin123");
        driver.findElement(By.xpath("//button[@type='submit']")).click();

        wait = new WebDriverWait(driver, Duration.ofSeconds(10));

        // Navigate to Admin page
        wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//a[@href='/web/index.php/admin/viewAdminModule']"))).click();
    }

    @When("I enter the following search criteria:")
    public void i_enter_the_following_search_criteria(DataTable dataTable) {
        // Convert DataTable to list of maps
        List<Map<String, String>> rows = dataTable.asMaps(String.class, String.class);

        for (Map<String, String> row : rows) {
            String username = row.get("Username");
            String userRole = row.get("User Role");
            String status = row.get("Status");

            // Enter username
            WebElement usernameInput = wait.until(ExpectedConditions.visibilityOfElementLocated(By.name("username")));
            usernameInput.clear();
            usernameInput.sendKeys(username);

            // Select User Role
            WebElement userRoleDropdown = driver.findElement(By.xpath("//label[text()='User Role']/following::div[1]//div[@role='listbox']"));
            userRoleDropdown.click();
            WebElement userRoleOption = wait.until(ExpectedConditions.elementToBeClickable(
                    By.xpath("//div[@role='option']//span[text()='" + userRole + "']")));
            userRoleOption.click();

            // Select Status
            WebElement statusDropdown = driver.findElement(By.xpath("//label[text()='Status']/following::div[1]//div[@role='listbox']"));
            statusDropdown.click();
            WebElement statusOption = wait.until(ExpectedConditions.elementToBeClickable(
                    By.xpath("//div[@role='option']//span[text()='" + status + "']")));
            statusOption.click();

            // Click Search button
            WebElement searchButton = driver.findElement(By.xpath("//button[contains(.,'Search')]"));
            searchButton.click();
        }
    }

    @Then("the search results should be displayed")
    public void the_search_results_should_be_displayed() {
        WebElement resultsTable = wait.until(ExpectedConditions.visibilityOfElementLocated(
                By.xpath("//div[@class='oxd-table-body']")));
        
        if (resultsTable.isDisplayed()) {
            System.out.println("Search results displayed successfully!");
        } else {
            System.out.println("No results found!");
        }

        driver.quit();
    }
}

package stepDefinitions;

import io.cucumber.java.en.*;

import java.time.Duration;

import static org.junit.Assert.assertEquals;

public class LeaveSteps {

    WebDriver driver;
    WebDriverWait wait;
    int initialLeaveBalance;
    int finalLeaveBalance;

    @Given("user is logged in and on the Leave page")
    public void user_is_logged_in_and_on_leave_page() {
        System.setProperty("webdriver.chrome.driver", "path/to/chromedriver");
        driver = new ChromeDriver();
        driver.manage().window().maximize();

        driver.get("https://opensource-demo.orangehrmlive.com/web/index.php/auth/login");
        driver.findElement(By.name("username")).sendKeys("Admin");
        driver.findElement(By.name("password")).sendKeys("admin123");
        driver.findElement(By.xpath("//button[@type='submit']")).click();

        wait = new WebDriverWait(driver, Duration.ofSeconds(10));

        // Navigate to Leave page
        wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//a[@href='/web/index.php/leave/viewLeaveModule']"))).click();
    }

    @When("I capture the initial leave balance")
    public void i_capture_the_initial_leave_balance() {
        WebElement balanceElement = wait.until(ExpectedConditions.visibilityOfElementLocated(
                By.xpath("//div[@class='oxd-text oxd-text--span']"))); // Replace with actual selector
        String balanceText = balanceElement.getText().trim();
        initialLeaveBalance = Integer.parseInt(balanceText);
        System.out.println("Initial Leave Balance: " + initialLeaveBalance);
    }

    @When("I apply leave for one day")
    public void i_apply_leave_for_one_day() {
        WebElement applyButton = wait.until(ExpectedConditions.elementToBeClickable(
                By.xpath("//button[contains(.,'Apply')]")));
        applyButton.click();

        WebElement leaveType = wait.until(ExpectedConditions.elementToBeClickable(
                By.xpath("//label[text()='Type']/following::div[1]//div[@role='listbox']")));
        leaveType.click();
        WebElement typeOption = wait.until(ExpectedConditions.elementToBeClickable(
                By.xpath("//div[@role='option']//span[text()='CAN']"))); // Example: Casual Leave
        typeOption.click();

        WebElement startDate = driver.findElement(By.xpath("//input[@placeholder='yyyy-mm-dd']"));
        startDate.clear();
        startDate.sendKeys("2026-05-10");

        WebElement endDate = driver.findElement(By.xpath("//input[@placeholder='yyyy-mm-dd' and contains(@class,'end-date')]"));
        endDate.clear();
        endDate.sendKeys("2026-05-10");

        WebElement submitButton = driver.findElement(By.xpath("//button[contains(.,'Submit')]"));
        submitButton.click();
    }

    @Then("I capture the success toast message")
    public void i_capture_the_success_toast_message() {
        WebElement toast = wait.until(ExpectedConditions.visibilityOfElementLocated(
                By.xpath("//div[contains(@class,'oxd-toast-content')]")));
        String message = toast.getText();
        System.out.println("Success Toast Message: " + message);
    }

    @Then("I verify the leave balance is reduced by one")
    public void i_verify_the_leave_balance_is_reduced_by_one() {
        WebElement balanceElement = wait.until(ExpectedConditions.visibilityOfElementLocated(
                By.xpath("//div[@class='oxd-text oxd-text--span']"))); // Replace with actual selector
        String balanceText = balanceElement.getText().trim();
        finalLeaveBalance = Integer.parseInt(balanceText);
        System.out.println("Final Leave Balance: " + finalLeaveBalance);

        assertEquals("Leave balance did not reduce by 1", initialLeaveBalance - 1, finalLeaveBalance);

        driver.quit();
    }
}

package stepDefinitions;

import io.cucumber.java.*;
import org.openqa.selenium.OutputType;
import org.openqa.selenium.TakesScreenshot;
import org.openqa.selenium.WebDriver;

import java.io.File;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Paths;

public class Hooks {

    public static WebDriver driver;

    @Before
    public void setUp() {
        System.setProperty("webdriver.chrome.driver", "path/to/chromedriver");
        driver = new org.openqa.selenium.chrome.ChromeDriver();
        driver.manage().window().maximize();
    }

    @After
    public void tearDown(Scenario scenario) {
        if (scenario.isFailed()) {
            try {
                TakesScreenshot ts = (TakesScreenshot) driver;
                byte[] screenshot = ts.getScreenshotAs(OutputType.BYTES);
                scenario.attach(screenshot, "image/png", "failure_screenshot");
                
                // Optional: save locally
                File screenshotFile = ts.getScreenshotAs(OutputType.FILE);
                Files.copy(screenshotFile.toPath(), Paths.get("target/screenshots/" + scenario.getName() + ".png"));
            } catch (IOException e) {
                e.printStackTrace();
            }
        }
        driver.quit();
    }
}
