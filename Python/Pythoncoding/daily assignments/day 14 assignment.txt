package pages;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;
import java.time.Duration;
public class BasePage {
    protected WebDriver driver;
    protected WebDriverWait wait;
    public BasePage(WebDriver driver) {
        this.driver = driver;
        this.wait = new WebDriverWait(driver, Duration.ofSeconds(10));
    }
    protected WebElement find(By locator) {
        return wait.until(ExpectedConditions.visibilityOfElementLocated(locator));
    }
    protected void click(By locator) {
        wait.until(ExpectedConditions.elementToBeClickable(locator)).click();
    }
    protected String getText(By locator) {
        return find(locator).getText();
    }
}

CategoryPage.java (Base for categories like Phones or Laptops)
package pages;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import java.util.List;
import java.util.stream.Collectors;
public class CategoryPage extends BasePage {
    private By productNames = By.cssSelector(".card-title a"); // All product links
    public CategoryPage(WebDriver driver) {
        super(driver);
    }
    public List<String> get_all_product_names() {
        List<WebElement> elements = driver.findElements(productNames);
        return elements.stream()
                .map(WebElement::getText)
                .collect(Collectors.toList());
    }
}

ProductDetailsPage.java
package pages;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;
import java.time.Duration;
import java.util.logging.Logger;
public class ProductDetailsPage extends BasePage {
    private By addToCartButton = By.xpath("//a[text()='Add to cart']");
    private static final Logger logger = Logger.getLogger(ProductDetailsPage.class.getName());
    public ProductDetailsPage(WebDriver driver) {
        super(driver);
    }
    public void add_product_to_cart() {
        click(addToCartButton);
        WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(10));
        wait.until(ExpectedConditions.alertIsPresent());
        driver.switchTo().alert().accept();
        logger.info("Product added to cart and alert accepted.");
        System.out.println("Product added to cart successfully!");
    }
}

package pages;
import io.qameta.allure.Allure;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import java.util.Map;
public class PurchaseModal extends BasePage {
    private By nameField = By.id("name");
    private By countryField = By.id("country");
    private By cityField = By.id("city");
    private By cardField = By.id("card");
    private By monthField = By.id("month");
    private By yearField = By.id("year");
    private By purchaseButton = By.xpath("//button[text()='Purchase']");
    private By successMessage = By.xpath("//h2[text()='Thank you for your purchase!']");
    public PurchaseModal(WebDriver driver) {
        super(driver);
    }
    public void fill_purchase_form(Map<String, String> data) {
        Allure.step("Filling Name field: " + data.get("Name"));
        find(nameField).sendKeys(data.get("Name"));
        Allure.step("Filling Country field: " + data.get("Country"));
        find(countryField).sendKeys(data.get("Country"));
        Allure.step("Filling City field: " + data.get("City"));
        find(cityField).sendKeys(data.get("City"));
        Allure.step("Filling Card field: " + data.get("Card"));
        find(cardField).sendKeys(data.get("Card"));
        Allure.step("Filling Month field: " + data.get("Month"));
        find(monthField).sendKeys(data.get("Month"));
        Allure.step("Filling Year field: " + data.get("Year"));
        find(yearField).sendKeys(data.get("Year"));
    }
    public void click_purchase() {
        find(purchaseButton).click();
    }
    public boolean verify_success_message() {
        return find(successMessage) != null;
    }
}

Exercise 5: Error Handling & Screenshots (Negative Test) The Task: Handle the "Login" modal. What to test: Click "Log in" in the header, enter a valid username but a wrong password, and click "Log in". POM Logic: Use `pytest` hooks in `conftest.py` to monitor this test. Screenshot Challenge: Intentionally write an assertion that fails (e.g., asserting the alert text is "Success"). The `conftest.py` must automatically capture a screenshot of the login modal and attach it to the Allure Report 





code:

login_page.py
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
class LoginPage:
    def __init__(self, driver):
        self.driver = driver
        self.username_input = (By.ID, "loginusername")
        self.password_input = (By.ID, "loginpassword")
        self.login_button = (By.XPATH, "//button[text()='Log in']")
    def open_login_modal(self):
        self.driver.find_element(By.ID, "login2").click()
    def login(self, username, password):
        wait = WebDriverWait(self.driver, 10)
        wait.until(EC.visibility_of_element_located(self.username_input)).send_keys(username)
        self.driver.find_element(*self.password_input).send_keys(password)
        self.driver.find_element(*self.login_button).click()
    def get_alert_text(self):
        wait = WebDriverWait(self.driver, 10)
        alert = wait.until(EC.alert_is_present())
        text = alert.text
        alert.accept()
        return text

import pytest

@pytest.mark.usefixtures("driver_init")
class TestLoginNegative:
    def test_wrong_password(self, driver):
        driver.get("https://www.demoblaze.com/index.html")
        login_page = LoginPage(driver)
        login_page.open_login_modal()
        login_page.login("valid_user", "wrong_password")
        alert_text = login_page.get_alert_text()
        # Intentional failing assertion to trigger screenshot
        assert alert_text == "Success", f"Expected 'Success', got '{alert_text}'"

conftest.py
import pytest
from selenium import webdriver
import allure
@pytest.fixture(scope="function")
def driver_init(request):
    driver = webdriver.Chrome()
    driver.maximize_window()
    request.cls.driver = driver
    yield driver
    driver.quit()
@pytest.hookimpl(hookwrapper=True)
def pytest_runtest_makereport(item, call):
    outcome = yield
    rep = outcome.get_result()
    if rep.when == "call" and rep.failed:
        driver = item.funcargs.get("driver")
        if driver:
            screenshot = driver.get_screenshot_as_png()
            allure.attach(screenshot, name="screenshot_on_failure", attachment_type=allure.attachment_type.
